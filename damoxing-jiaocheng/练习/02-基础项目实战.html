# chat.py
from openai import OpenAI

# Ollama 兼容 OpenAI 接口，所以直接用 openai 库就能调本地模型
client = OpenAI(base_url="http://localhost:11434/v1", api_key="ollama")

resp = client.chat.completions.create(
    model="qwen2.5:1.5b",
    messages=[{"role": "user", "content": "用一句话解释什么是内存"}],

项目 2：命令行翻译助手（1 小时）

目标：做一个"中译英"工具，理解 System Prompt 的作用。

# translator.py
from openai import OpenAI

client = OpenAI(base_url="http://localhost:11434/v1", api_key="ollama")

SYSTEM = "你是一个专业翻译。把用户输入翻译成英语，只输出译文，不要解释。"

def translate(text):
    resp = client.chat.completions.create(
        model="qwen2.5:1.5b",
        messages=[
            {"role": "system", "content": SYSTEM},
            {"role": "user", "content": text},
        ],
    )
    return resp.choices[0].message.content

if __name__ == "__main__":
    while True:
        text = input("请输入中文（q 退出）: ")
        if text.lower() == "q":
            break
        print("翻译:", translate(text))

学到的知识点：System Prompt 控制行为、角色设定。

扩展练习：
- 加"翻译成日语/法语"选项
- 加"润色文案"模式（换 SYSTEM 就行）

项目 3：本地知识库问答（RAG 入门，半天）

目标：把你的笔记/文档做成 AI 能查的知识库。这是最实用、面试最爱问的项目。

原理回顾：文档切片 → 向量化 → 存向量库 → 检索 → 喂给模型。

# rag_demo.py
import chromadb
from openai import OpenAI

client = OpenAI(base_url="http://localhost:11434/v1", api_key="ollama")
vector_client = chromadb.Client()
col = vector_client.create_collection("my_notes")

# 第 1 步：把文档切片存进向量库（这里用你自己的 txt 笔记）
with open("我的笔记.txt", "r", encoding="utf-8") as f:
    text = f.read()
chunks = [text[i:i+500] for i in range(0, len(text), 500)]  # 500 字一段

# 第 2 步：用 Embedding 模型向量化（先拉取：ollama pull nomic-embed-text）
from chromadb.utils.embedding_functions import OllamaEmbeddingFunction
ef = OllamaEmbeddingFunction(url="http://localhost:11434/api/embeddings", model_name="nomic-embed-text")

for i, chunk in enumerate(chunks):
    col.add(ids=[str(i)], documents=[chunk], embeddings=ef([chunk]))

# 第 3 步：提问时先检索再回答
question = "我的笔记里提到了什么主题？"
results = col.query(query_embeddings=ef([question]), n_results=3)
context = "\n".join(results["documents"][0])

resp = client.chat.completions.create(
    model="qwen2.5:1.5b",
    messages=[{"role": "system", "content": "根据资料回答问题，资料里没有就说不知道。"},
              {"role": "user", "content": f"资料：\n{context}\n\n问题：{question}"}],
)
print(resp.choices[0].message.content)

先拉 embedding 模型：

ollama pull nomic-embed-text

学到的知识点：Embedding、向量数据库、RAG 全流程、上下文拼接。

扩展练习：
- 换你的真实笔记/工作文档
- 问"笔记里没写"的问题，验证"说不知道"的效果

项目 4：AI 网页助手（半天）

目标：做一个本地网页聊天界面，体验"产品化"。

# 装 Web 框架
pip install fastapi uvicorn

# web_chat.py
from fastapi import FastAPI
from pydantic import BaseModel
from openai import OpenAI

app = FastAPI()
client = OpenAI(base_url="http://localhost:11434/v1", api_key="ollama")

class Msg(BaseModel):
    content: str

@app.post("/chat")
def chat(msg: Msg):
    resp = client.chat.completions.create(
        model="qwen2.5:1.5b",
        messages=[{"role": "user", "content": msg.content}],
    )
    return {"reply": resp.choices[0].message.content}

# 启动：uvicorn web_chat:app --port 8000
# 浏览器访问 http://localhost:8000/docs 就能测试接口

学到的知识点：API 服务、前后端分离思想、模型部署。

扩展练习：
- 加一个简单 HTML 前端页面

项目 5：LoRA 微调小模型（进阶，1~2 天）

目标：用你自己的数据"教"模型新东西。这是从"用户"到"开发者"的跨越。

在你的配置上，推荐微调 0.5B~1.5B 的小模型（LoRA 补丁很小，训练快）。

# 用 mlx-lm 微调（Apple Silicon 最快方案）
pip install mlx-lm

准备训练数据 train.jsonl（每行一条）：

{"text": "用户：今天天气怎么样？\n助手：今天晴转多云，气温 25 度，适合出门。"}
{"text": "用户：推荐一家火锅店\n助手：推荐你试试川味老灶，毛肚和鸭肠是一绝。"}

训练（在终端执行，约 10~20 分钟）：

python -m mlx_lm.lora \
  --model Qwen/Qwen2.5-0.5B-Instruct-4bit \
  --train \
  --data ./train.jsonl \
  --iters 200 \
  --batch-size 4 \
  --learning-rate 1e-5 \
  --adapter-path ./my-adapter

测试微调效果：

python -m mlx_lm.generate \
  --model Qwen/Qwen2.5-0.5B-Instruct-4bit \
  --adapter-path ./my-adapter \
  --prompt "推荐一家火锅店"

学到的知识点：数据准备、LoRA 原理落地、训练超参数（学习率/迭代次数）。

扩展练习：
- 收集 100 条你自己的问答数据（比如你工作领域的话术）
- 对比微调前后回答差异
- 试试 QLoRA（量化 + LoRA 组合）

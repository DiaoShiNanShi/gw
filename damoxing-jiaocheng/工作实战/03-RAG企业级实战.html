1. 为什么企业首选 RAG？（先搞清楚动机）

需求 RAG 微调
公司内部资料（制度/产品手册） ✅ 立刻能用 ❌ 训练成本高
知识实时更新（新政策/新价格） ✅ 改文档即可 ❌ 要重新训练

不需要改模型行为/语气 ✅ ⚠️ 杀鸡用牛刀

一句话：RAG 解决"模型不知道的事"，微调解决"模型不会做的事"。 企业 80% 的需求是前者。

4. 关键细节：切分策略（最影响效果的一步）

为什么切分很重要？ 检索是按"片段"找的。切太碎 → 语义不完整；切太长 → 混入噪音。

反例：把整份 50 页的合同当一个片段，检索时"年假 15 天"和"离职赔偿"全混在一起，模型分不清。
正例：按章节/段落切，每段 300~500 字，段与段之间保留少量重叠（overlap 50~100 字），防止关键句被切断。

def split_document(text, chunk_size=400, overlap=80):
    """按字符数切分，带重叠"""
    chunks = []
    for i in range(0, len(text), chunk_size - overlap):
        chunks.append(text[i : i + chunk_size])
    return chunks

# 进阶：按 Markdown 标题切（保持章节完整性）
import re
def split_by_heading(text):
    parts = re.split(r'(?m)^#{1,3} ', text)  # 按 # 标题切
    return [p.strip() for p in parts if len(p.strip()) > 50]

企业级切分建议：先按结构切（章节），再对超长章节按字符切 + 重叠。

5. 完整可运行的 RAG 系统（本地版，你的 Mac 能跑）

"""
本地 RAG 系统：Ollama 提供模型 + ChromaDB 存向量
安装：pip install chromadb  &&  ollama pull qwen2.5:1.5b && ollama pull nomic-embed-text
"""
from openai import OpenAI
import chromadb
from chromadb.utils.embedding_functions import OllamaEmbeddingFunction

llm = OpenAI(base_url="http://localhost:11434/v1", api_key="ollama")
ef = OllamaEmbeddingFunction(
    url="http://localhost:11434/api/embeddings", model_name="nomic-embed-text"
)
db = chromadb.PersistentClient(path="./kb")   # 持久化到本地文件夹
col = db.get_or_create_collection("docs", embedding_function=ef)

def ingest(text, doc_id):
    """入库：切分 → 向量化 → 存储"""
    chunks = [text[i:i+400] for i in range(0, len(text), 320)]  # 400字+80重叠
    col.add(
        ids=[f"{doc_id}_{i}" for i in range(len(chunks))],
        documents=chunks,
        metadatas=[{"doc_id": doc_id} for _ in chunks],
    )

def query(question, top_k=3):
    """检索 + 生成"""
    hits = col.query(query_texts=[question], n_results=top_k)
    context = "\n\n".join(hits["documents"][0])
    resp = llm.chat.completions.create(
        model="qwen2.5:1.5b",
        messages=[
            {"role": "system", "content": "你是知识库助手。只根据提供的资料回答；资料没有就说'资料中未提及'。回答末尾标注资料编号[1][2]。"},
            {"role": "user", "content": f"资料：\n{context}\n\n问题：{question}"},
        ],
    )
    return resp.choices[0].message.content

6. 企业级进阶（让 RAG 从"能用"到"好用"）

进阶 1：混合检索（向量 + 关键词）

痛点：向量检索对"精确匹配"不敏感。搜"iPhone 15 价格"，文档里写的是"iPhone 十五代售价"，向量可能匹配不到。

解法：向量检索 + BM25 关键词检索，结果合并（RRF 融合算法），双保险。

# 用 Elasticsearch 或 rank_bm25 做关键词检索
from rank_bm25 import BM25Okapi

def hybrid_search(question, top_k=5):
    # 向量结果
    vec_hits = col.query(query_texts=[question], n_results=top_k)["documents"][0]
    # 关键词结果（本地 BM25）

    bm25 = BM25Okapi(tokenized_docs)
    kw_hits = [all_docs[i] for i in bm25.get_top_n(question.split(), all_docs, n=top_k)]
    # 合并去重（简单版：先向量后关键词）
    return list(dict.fromkeys(vec_hits + kw_hits))[:top_k]

进阶 2：重排序（Rerank）——检索质量翻倍的关键

痛点：向量库返回的 Top-5 里，可能只有 2 条真正相关。直接全喂给模型，会混淆。

解法：先用便宜的方式召回 20 条，再用专门的 Rerank 模型精排，只留 Top-3 喂给 LLM。

# 使用 bge-reranker（本地跑）或 API 版
from FlagEmbedding import FlagReranker
reranker = FlagReranker("BAAI/bge-reranker-base", use_fp16=True)

def rerank(question, candidates, top_k=3):
    scores = reranker.compute_score([[question, c] for c in candidates])
    ranked = sorted(zip(candidates, scores), key=lambda x: x[1], reverse=True)
    return [c for c, s in ranked[:top_k]]

进阶 3：查询改写（先理解再检索）

痛点：用户问"它什么时候到期？"——"它"指代不明，直接检索效果差。

解法：先用 LLM 把口语问题改写为检索友好的完整问题。

def rewrite_query(history, raw_question):
    resp = llm.chat.completions.create(
        model="qwen2.5:1.5b",
        messages=[{"role": "system", "content": "根据对话历史，把用户问题改写为独立、完整、适合检索的问题。只输出改写结果。"},
                  {"role": "user", "content": f"历史：{history}\n问题：{raw_question}"}],
    )
    return resp.choices[0].message.content

进阶 4：引文溯源（企业必做）

要求模型输出带来源编号，并把编号映射到原始文档链接。这样用户能点开原文核对——这是企业知识库敢用的前提（出了错能追责）。

8. 文档解析（PDF/Word/Excel 怎么进知识库）

# PDF 解析（pip install pypdf）
from pypdf import PdfReader
def read_pdf(path):
    reader = PdfReader(path)
    return "\n".join(page.extract_text() for page in reader.pages)

# Word 解析（pip install python-docx）
from docx import Document
def read_docx(path):
    doc = Document(path)
    return "\n".join(p.text for p in doc.paragraphs)

# 表格类（pip install openpyxl）
from openpyxl import load_workbook
def read_xlsx(path):
    wb = load_workbook(path)
    rows = []
    for ws in wb.worksheets:
        for row in ws.iter_rows(values_only=True):
            rows.append(" | ".join(str(c) for c in row if c is not None))
    return "\n".join(rows)

企业注意：扫描版 PDF 是图片，要 OCR（如 PaddleOCR）。这经常是知识库项目里最费时间的环节。
